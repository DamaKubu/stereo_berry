3. Motion detection (cheap and effective)
def threshold_and_find_blobs(D):
    D_blur = gaussian_blur(D, sigma=1.0)
    mask = D_blur > MOTION_THRESH
    blobs = connected_components(mask)
    return blobs

def filter_blobs(blobs):
    dets = []
    for b in blobs:
        if MIN_AREA < b.area < MAX_AREA:
            if ASPECT_MIN < b.aspect_ratio < ASPECT_MAX:
                dets.append(
                    Detection(
                        bbox=b.bbox,
                        centroid=b.centroid,
                        area=b.area
                    )
                )
    return dets

4. Tracking (SORT-style, simplified)
def predict_tracks(tracks):
    for trk in tracks:
        trk.kf.predict()
    return tracks

def associate_detections(tracks, detections):
    cost = build_cost_matrix(tracks, detections)  # euclidean distance
    matches, unmatched_tracks, unmatched_dets = hungarian(cost)
    return matches, unmatched_tracks, unmatched_dets

def update_tracks(tracks, detections):
    for (trk, det) in matches:
        trk.kf.update(det.centroid)
        trk.hits += 1
        trk.miss = 0
        trk.age += 1

        trk.history_pos.append(det.centroid)
        trk.history_area.append(det.area)
        trk.history_vel.append(trk.kf.velocity())

    for trk in unmatched_tracks:
        trk.miss += 1
        trk.age += 1

    for det in unmatched_dets:
        tracks.append(init_new_track(det))

def prune_dead_tracks(tracks):
    return [t for t in tracks if t.miss < MAX_MISSES]

5. Bearing estimation (core geometry)
def update_bearing(track):
    u, v = track.kf.position()

    # Undistort single point
    u_ud, v_ud = undistort_point(u, v, K, dist)

    # Camera bearing
    x = (u_ud - cx) / fx
    y = (v_ud - cy) / fy
    b_cam = normalize([x, y, 1.0])

    # World bearing
    b_world = R_cam_to_world @ b_cam

    # Temporal smoothing
    track.bearing_world = ema(track.bearing_world, b_world, alpha=0.2)

6. Motion-based drone vs bird discrimination
Feature extraction (per track)
def compute_motion_features(track):
    v = track.history_vel[-N:]
    a = diff(v)

    vel_var = variance(norm(v))
    acc_var = variance(norm(a))

    area = track.history_area[-N:]
    area_var = std(area) / mean(area)

    return vel_var, acc_var, area_var

Rule-based classifier (surprisingly strong)
def update_class_score(track):
    vel_var, acc_var, area_var = compute_motion_features(track)

    score = 0.0

    if vel_var < VEL_VAR_THRESH:
        score += 0.4
    if acc_var < ACC_VAR_THRESH:
        score += 0.4
    if area_var < AREA_VAR_THRESH:
        score += 0.2

    track.class_score = ema(track.class_score, score, alpha=0.3)
